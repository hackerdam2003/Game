Model ka bounding box (total size) pata karo const box = new THREE.Box3().setFromObject(houseModel); const center = box.getCenter(new THREE.Vector3()); const size = box.getSize(new THREE.Vector3()); // 2. Model ko screen ke ekdum center me laao houseModel.position.x = -center.x; houseModel.position.y = -center.y; houseModel.position.z = -center.z; // 3.
